**Goal:** 实现 Python 包 `vc_eval`：用留出段的老师融合标注作代理参照，量化 `vc_redact` 打码视频的最终覆盖率、时间覆盖率、输出完整性、C++ / Python 推理一致性和各阶段速度，按检测频率扫描，并生成本地抽查页与汇总表。

```python
"""Coverage geometry and statistics (proxy-validation spec §3.3, §6; main spec §8.1)."""

from __future__ import annotations

import math
from dataclasses import dataclass

import numpy as np

from vc_label.contracts import Box

FACE_MIN_4K = 24.0  # must-mask face height at 2160 px
PLATE_MIN_4K = 48.0  # must-mask plate width at 2160 px
BIN_EDGES_4K = (24.0, 48.0, 96.0, 192.0)
COVERED_FRACTION = 0.8
GRID_COLS = 6
GRID_ROWS = 4
HEATMAP_MIN_TARGETS = 30
CLASS_PASS = 0.99
CELL_PASS = 0.97


def target_size(box: Box) -> float:
    """Faces are sized by height, plates by width."""
    return box.height if box.cls == "face" else box.width


def must_mask(box: Box, frame_height: int) -> bool:
    limit = FACE_MIN_4K if box.cls == "face" else PLATE_MIN_4K
    return target_size(box) >= limit * frame_height / 2160


def _edges(frame_height: int) -> list[float]:
    return [e * frame_height / 2160 for e in BIN_EDGES_4K]


def size_bin_labels(frame_height: int) -> list[str]:
    e = _edges(frame_height)
    return [f"<{e[0]:g}"] + [f"{e[i]:g}-{e[i + 1]:g}" for i in range(len(e) - 1)] + [f">{e[-1]:g}"]


def size_bin(box: Box, frame_height: int) -> str:
    edges = _edges(frame_height)
    labels = size_bin_labels(frame_height)
    size = target_size(box)
    for i, edge in enumerate(edges):
        if size < edge:
            return labels[i]
    return labels[-1]


def covered_fraction(target: Box, masks: list[Box]) -> float:
    """Exact share of the target's area under the union of the masks (any class)."""
    if target.area <= 0:
        return 0.0
    parts = [
        (max(m.x1, target.x1), max(m.y1, target.y1), min(m.x2, target.x2), min(m.y2, target.y2))
        for m in masks
    ]
    parts = [p for p in parts if p[2] > p[0] and p[3] > p[1]]
    if not parts:
        return 0.0
    xs = sorted({target.x1, target.x2, *(p[0] for p in parts), *(p[2] for p in parts)})
    ys = sorted({target.y1, target.y2, *(p[1] for p in parts), *(p[3] for p in parts)})
    covered = 0.0
    for xa, xb in zip(xs, xs[1:]):
        cx = (xa + xb) / 2
        for ya, yb in zip(ys, ys[1:]):
            cy = (ya + yb) / 2
            if any(p[0] <= cx < p[2] and p[1] <= cy < p[3] for p in parts):
                covered += (xb - xa) * (yb - ya)
    return covered / target.area


def _pixels(box: Box, width: int, height: int) -> tuple[slice, slice]:
    rows = slice(max(0, math.floor(box.y1)), min(height, math.ceil(box.y2)))
    cols = slice(max(0, math.floor(box.x1)), min(width, math.ceil(box.x2)))
    return rows, cols


def stray_mask_fraction(masks: list[Box], targets: list[Box], width: int, height: int) -> float:
    """Share of the frame covered by masks but by no target box (the false-positive area)."""
    if not masks:
        return 0.0
    grid = np.zeros((height, width), dtype=bool)
    for m in masks:
        grid[_pixels(m, width, height)] = True
    for t in targets:
        grid[_pixels(t, width, height)] = False
    return float(grid.sum()) / (width * height)


def heat_cell(box: Box, width: int, height: int) -> tuple[int, int]:
    cx, cy = box.center
    col = min(GRID_COLS - 1, max(0, int(cx / width * GRID_COLS)))
    row = min(GRID_ROWS - 1, max(0, int(cy / height * GRID_ROWS)))
    return row, col


def wilson_interval(hits: int, total: int, z: float = 1.96) -> tuple[float, float]:
    if total == 0:
        return (0.0, 1.0)
    p = hits / total
    denom = 1 + z * z / total
    centre = (p + z * z / (2 * total)) / denom
    half = z * math.sqrt(p * (1 - p) / total + z * z / (4 * total * total)) / denom
    return (max(0.0, centre - half), min(1.0, centre + half))


@dataclass(frozen=True)
class Rate:
    hits: int = 0
    total: int = 0

    @property
    def value(self) -> float:
        return self.hits / self.total if self.total else float("nan")

    @property
    def interval(self) -> tuple[float, float]:
        return wilson_interval(self.hits, self.total)

    def add(self, hit: bool) -> Rate:
        return Rate(self.hits + int(hit), self.total + 1)
```

```python
import csv
import json

import cv2
import numpy as np
import pytest

from vc_label.contracts import Box, Detection, write_jsonl
from vc_eval.coverage import evaluate_coverage
from vc_eval.records import read_masks
from vc_eval.temporal import evaluate_temporal

W, H = 1920, 1080


def _frames(root, stem, frames):
    d = root / "frames" / stem
    d.mkdir(parents=True)
    with (d / "frames.csv").open("w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["video", "frame", "pts_ms", "reason", "motion_score"])
        for fr in frames:
            w.writerow([f"{stem}.mp4", fr, fr * 16.7, "base", 0.0])
            cv2.imwrite(str(d / f"{fr:06d}.jpg"), np.zeros((H, W, 3), np.uint8))
    return d


def _labels(root, stem, boxes_by_frame):
    d = root / "labels" / stem
    d.mkdir(parents=True)
    (d / "processed.txt").write_text("".join(f"{fr}\n" for fr in boxes_by_frame))
    write_jsonl(d / "merged.jsonl", [Detection(f"{stem}.mp4", fr, fr * 16.7, b) for fr, bs in boxes_by_frame.items() for b in bs])
    return d


def _masks(path, video, masks_by_frame, frames):
    with path.open("w") as f:
        for fr in frames:
            boxes = [{"cls": b.cls, "x1": b.x1, "y1": b.y1, "x2": b.x2, "y2": b.y2, "origin": "detected"} for b in masks_by_frame.get(fr, [])]
            f.write(json.dumps({"video": video, "frame": fr, "boxes": boxes}) + "\n")
    return path
```

```python
"""Readers for vc_redact outputs: masks.jsonl (one line per frame) and detections.jsonl."""

from __future__ import annotations

import json
from collections import defaultdict
from pathlib import Path

from vc_label.contracts import Box, read_jsonl


def read_masks(path: Path) -> tuple[set[str], dict[int, list[Box]]]:
    """(video names, frame -> mask boxes); each Box.src holds the mask origin."""
    videos: set[str] = set()
    by_frame: dict[int, list[Box]] = {}
    with path.open(encoding="utf-8") as f:
        for n, line in enumerate(f, 1):
            if not line.strip():
                continue
            try:
                rec = json.loads(line)
                videos.add(rec["video"])
                by_frame[int(rec["frame"])] = [
                    Box(b["cls"], float(b["x1"]), float(b["y1"]), float(b["x2"]), float(b["y2"]), src=b["origin"])
                    for b in rec["boxes"]
                ]
            except (KeyError, TypeError, ValueError) as e:
                raise ValueError(f"{path} line {n}: bad mask record ({e})") from None
    return videos, by_frame


def detections_by_frame(path: Path) -> tuple[set[str], dict[int, list[Box]]]:
    videos: set[str] = set()
    by_frame: dict[int, list[Box]] = defaultdict(list)
    for d in read_jsonl(path):
        videos.add(d.video)
        by_frame[d.frame].append(d.box)
    return videos, dict(by_frame)
```

```python
"""Final coverage of the redacted video against the held-out teacher labels (proxy-validation spec §6.1)."""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

import numpy as np

from vc_label.contracts import LABEL_CLASSES, Box, read_frames_csv
from vc_label.pipeline import MERGED, check_labels_match_frames, read_processed
from vc_label.review import image_size
from vc_label.rules import TEMPORAL_SOURCE
from vc_eval.metrics import (
    CELL_PASS,
    CLASS_PASS,
    COVERED_FRACTION,
    GRID_COLS,
    GRID_ROWS,
    HEATMAP_MIN_TARGETS,
    Rate,
    covered_fraction,
    heat_cell,
    must_mask,
    size_bin,
    size_bin_labels,
    stray_mask_fraction,
)
from vc_eval.records import detections_by_frame, read_masks


@dataclass(frozen=True)
class Miss:
    frame: int
    box: Box
    covered: float


@dataclass
class ClassCoverage:
    rate: Rate
    bins: dict[str, Rate]
    heatmap: list[list[Rate]]
    temporal_fill: Rate
    best_effort: Rate


def _empty_class(frame_height: int) -> ClassCoverage:
    return ClassCoverage(
        rate=Rate(),
        bins={label: Rate() for label in size_bin_labels(frame_height)},
        heatmap=[[Rate() for _ in range(GRID_COLS)] for _ in range(GRID_ROWS)],
        temporal_fill=Rate(),
        best_effort=Rate(),
    )


@dataclass
class CoverageReport:
    video: str
    frames: int
    width: int
    height: int
    classes: dict[str, ClassCoverage]
    misses: list[Miss] = field(default_factory=list)
    stray_max: float = 0.0
    stray_mean: float = 0.0
    stray_p95: float = 0.0

    def class_passed(self, cls: str) -> bool:
        rate = self.classes[cls].rate
        return rate.total > 0 and rate.value >= CLASS_PASS

    def weakest_cell(self) -> tuple[str, int, int, Rate] | None:
        cells = [
            (cls, r, c, rate)
            for cls, cc in self.classes.items()
            for r, row in enumerate(cc.heatmap)
            for c, rate in enumerate(row)
            if rate.total >= HEATMAP_MIN_TARGETS
        ]
        return min(cells, key=lambda x: x[3].value, default=None)

    def heatmap_passed(self) -> bool:
        weakest = self.weakest_cell()
        return weakest is None or weakest[3].value >= CELL_PASS

    def passed(self) -> bool:
        return all(self.class_passed(c) for c in LABEL_CLASSES) and self.heatmap_passed()


def evaluate_coverage(frames_dir: Path, labels_dir: Path, masks_path: Path) -> CoverageReport:
    refs = read_frames_csv(frames_dir)
    check_labels_match_frames(frames_dir, labels_dir, refs)
    done = read_processed(labels_dir)
    used = [r for r in refs if r.frame in done]
    if not used:
        raise ValueError(f"{labels_dir} has no labelled frames")
    video = used[0].video
    mask_videos, masks = read_masks(masks_path)
    if mask_videos - {video}:
        raise ValueError(f"{masks_path} is for {', '.join(sorted(mask_videos))}, not {video}")
    _, targets = detections_by_frame(labels_dir / MERGED)
    width, height = image_size(used[0].image)
    report = CoverageReport(video, len(used), width, height, {c: _empty_class(height) for c in LABEL_CLASSES})
    stray = []
    for ref in used:
        if ref.frame not in masks:
            raise ValueError(f"{masks_path} has no line for frame {ref.frame}")
        frame_targets = [b for b in targets.get(ref.frame, []) if b.cls in LABEL_CLASSES]
        frame_masks = masks[ref.frame]
        stray.append(stray_mask_fraction(frame_masks, frame_targets, width, height))
        for t in frame_targets:
            cc = report.classes[t.cls]
            covered = covered_fraction(t, frame_masks)
            hit = covered >= COVERED_FRACTION
            if not must_mask(t, height):
                cc.best_effort = cc.best_effort.add(hit)
                continue
            cc.rate = cc.rate.add(hit)
            label = size_bin(t, height)
            cc.bins[label] = cc.bins[label].add(hit)
            r, c = heat_cell(t, width, height)
            cc.heatmap[r][c] = cc.heatmap[r][c].add(hit)
            if t.src == TEMPORAL_SOURCE:
                cc.temporal_fill = cc.temporal_fill.add(hit)
            if not hit:
                report.misses.append(Miss(ref.frame, t, covered))
    report.stray_max = float(max(stray))
    report.stray_mean = float(np.mean(stray))
    report.stray_p95 = float(np.percentile(stray, 95))
    return report
```

```python
"""Temporal coverage: sparse keyframes + BoxTimeline against per-frame detection (proxy-validation spec §6.2)."""

from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

from vc_label.contracts import LABEL_CLASSES
from vc_eval.coverage import Miss
from vc_eval.metrics import COVERED_FRACTION, Rate, covered_fraction, must_mask
from vc_eval.records import detections_by_frame, read_masks


@dataclass
class TemporalReport:
    video: str
    frames: int
    classes: dict[str, Rate]
    misses: list[Miss] = field(default_factory=list)


def evaluate_temporal(dense_detections: Path, sparse_masks: Path, frame_height: int) -> TemporalReport:
    """Every must-mask box of the --detect-every-frame run must be >= 80 % under that frame's sparse masks."""
    dense_videos, dense = detections_by_frame(dense_detections)
    mask_videos, masks = read_masks(sparse_masks)
    if dense_videos and dense_videos != mask_videos:
        raise ValueError(
            f"{dense_detections} ({', '.join(sorted(dense_videos))}) and {sparse_masks} "
            f"({', '.join(sorted(mask_videos))}) are for different videos"
        )
    report = TemporalReport(next(iter(mask_videos), ""), len(masks), {c: Rate() for c in LABEL_CLASSES})
    for frame in sorted(dense):
        if frame not in masks:
            raise ValueError(f"{sparse_masks} has no line for frame {frame}")
        for box in dense[frame]:
            if box.cls not in LABEL_CLASSES or not must_mask(box, frame_height):
                continue
            covered = covered_fraction(box, masks[frame])
            hit = covered >= COVERED_FRACTION
            report.classes[box.cls] = report.classes[box.cls].add(hit)
            if not hit:
                report.misses.append(Miss(frame, box, covered))
    return report
```

```python
"""Output integrity of a redacted video against its source (proxy-validation spec §6.4)."""

from __future__ import annotations

import json
import shutil
import subprocess
from dataclasses import dataclass, field
from pathlib import Path

FFPROBE = shutil.which("ffprobe") or "/opt/homebrew/bin/ffprobe"
FFMPEG = shutil.which("ffmpeg") or "/opt/homebrew/bin/ffmpeg"


@dataclass(frozen=True)
class StreamInfo:
    codec: str
    frames: int | None
    duration: float | None
    start: float


@dataclass(frozen=True)
class MediaInfo:
    video: StreamInfo
    audio: StreamInfo | None
    fps: float
    duration: float


def _number(value, cast=float):
    try:
        return cast(value)
    except (TypeError, ValueError):
        return None


def _rate(text: str) -> float:
    num, _, den = (text or "0/1").partition("/")
    return float(num) / float(den or 1) if float(den or 1) else 0.0


def _stream(s: dict) -> StreamInfo:
    return StreamInfo(
        codec=s.get("codec_name", ""),
        frames=_number(s.get("nb_read_frames"), int),
        duration=_number(s.get("duration")),
        start=_number(s.get("start_time")) or 0.0,
    )


def probe(path: Path) -> MediaInfo:
    """Streams of a media file; decodes the video to count frames (ffprobe -count_frames)."""
    out = subprocess.run(
        [FFPROBE, "-v", "error", "-count_frames", "-show_streams", "-show_format", "-of", "json", str(path)],
        capture_output=True, text=True, check=False,
    )
    if out.returncode != 0:
        raise ValueError(f"ffprobe failed on {path}: {out.stderr.strip()}")
    data = json.loads(out.stdout)
    streams = data.get("streams", [])
    video = next((s for s in streams if s.get("codec_type") == "video"), None)
    if video is None:
        raise ValueError(f"ffprobe found no video stream in {path}")
    audio = next((s for s in streams if s.get("codec_type") == "audio"), None)
    return MediaInfo(
        video=_stream(video),
        audio=_stream(audio) if audio else None,
        fps=_rate(video.get("avg_frame_rate")),
        duration=_number(data.get("format", {}).get("duration")) or 0.0,
    )


@dataclass
class IntegrityReport:
    checks: dict[str, bool] = field(default_factory=dict)
    details: dict[str, str] = field(default_factory=dict)

    @property
    def passed(self) -> bool:
        return bool(self.checks) and all(self.checks.values())


def check_integrity(source: Path, redacted: Path) -> IntegrityReport:
    s, r = probe(source), probe(redacted)
    frame = 1.0 / s.fps if s.fps > 0 else 1.0 / 30
    report = IntegrityReport()
    report.checks["decodable"] = bool(r.video.frames)
    report.details["decodable"] = f"{r.video.frames} frames decoded"
    report.checks["frame_count"] = (
        r.video.frames is not None and s.video.frames is not None and abs(r.video.frames - s.video.frames) <= 1
    )
    report.details["frame_count"] = f"source {s.video.frames}, redacted {r.video.frames}"
    report.checks["duration"] = abs(r.duration - s.duration) <= frame
    report.details["duration"] = f"source {s.duration:.3f} s, redacted {r.duration:.3f} s"
    if s.audio is None:
        report.checks["audio_kept"] = True
        report.checks["av_sync"] = True
        report.details["audio_kept"] = report.details["av_sync"] = "source has no audio"
        return report
    a = r.audio
    report.checks["audio_kept"] = (
        a is not None and a.codec == s.audio.codec and abs((a.duration or 0.0) - (s.audio.duration or 0.0)) <= frame
    )
    report.details["audio_kept"] = (
        f"source {s.audio.codec} {s.audio.duration} s, redacted "
        + (f"{a.codec} {a.duration} s" if a else "none")
    )
    if a is None:
        report.checks["av_sync"] = False
        report.details["av_sync"] = "redacted has no audio"
        return report
    shift = (a.start - r.video.start) - (s.audio.start - s.video.start)
    report.checks["av_sync"] = abs(shift) <= frame
    report.details["av_sync"] = f"audio offset changed by {shift * 1000:.1f} ms (limit {frame * 1000:.1f} ms)"
    return report
```

```python
    max_diff_px: float
    unmatched_confident: int
    unmatched_low: int

    def passed(self, tol_px: float = 1.0) -> bool:
        return self.unmatched_confident == 0 and self.max_diff_px < tol_px


def python_detections(model_dir: Path, refs: list[FrameRef], score: float = 0.15, iou_threshold: float = 0.5) -> dict[int, list[Box]]:
    import onnxruntime as ort

    meta = json.loads((model_dir / "model.json").read_text(encoding="utf-8"))
    session = ort.InferenceSession(str(model_dir / "model.onnx"), providers=["CPUExecutionProvider"])
    name = session.get_inputs()[0].name
    width_in, height_in = meta["input"]["width"], meta["input"]["height"]
    out: dict[int, list[Box]] = {}
    for ref in refs:
        bgr = cv2.imread(str(ref.image))
        if bgr is None:
            raise ValueError(f"cannot read {ref.image}")
        arr, r = letterbox(bgr, width_in, height_in)
        pred = session.run(None, {name: arr})[0]
        out[ref.frame] = decode_yolov8(pred, r, meta["classes"], bgr.shape[1], bgr.shape[0], score, iou_threshold)
    return out


def compare_detections(cpp: dict[int, list[Box]], py: dict[int, list[Box]]) -> ConsistencyReport:
    frames = sorted(set(cpp) | set(py))
    boxes = matched = confident = low = 0
    worst = 0.0
    for frame in frames:
        remaining = list(py.get(frame, []))
        for box in cpp.get(frame, []):
            boxes += 1
            same = [o for o in remaining if o.cls == box.cls]
            match = max(same, key=lambda o: iou(o, box), default=None)
            if match is None or iou(match, box) < MATCH_IOU:
                confident, low = (confident + 1, low) if box.score >= CONFIDENT else (confident, low + 1)
                continue
            remaining.remove(match)
            matched += 1
            worst = max(worst, *(abs(p - q) for p, q in zip((box.x1, box.y1, box.x2, box.y2), (match.x1, match.y1, match.x2, match.y2))))
        for box in remaining:
            confident, low = (confident + 1, low) if box.score >= CONFIDENT else (confident, low + 1)
    return ConsistencyReport(len(frames), boxes, matched, worst, confident, low)


def run_vc_detect(vc_detect: Path, model_dir: Path, refs: list[FrameRef], work_dir: Path, runner=subprocess.run) -> dict[int, list[Box]]:
    """Runs the C++ detector on `refs` through a scratch frames dir holding only those frames."""
    frames_dir = work_dir / "frames"
    frames_dir.mkdir(parents=True, exist_ok=True)
    with (frames_dir / "frames.csv").open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["video", "frame", "pts_ms", "reason", "motion_score"])
        for ref in refs:
            link = frames_dir / f"{ref.frame:06d}.jpg"
            if link.is_symlink() or link.exists():
                link.unlink()
            os.symlink(ref.image.resolve(), link)
            writer.writerow([ref.video, ref.frame, ref.pts_ms, "base", 0.0])
    out = work_dir / "cpp_detections.jsonl"
    runner(
        [str(vc_detect), "--model", str(model_dir), "--frames", str(frames_dir), "--out", str(out), "--ep", "cpu"],
        check=True, capture_output=True, text=True,
    )
    return detections_by_frame(out)[1]
```

```python
"""Local static spot-check page for the user (proxy-validation spec §6.5): no external resources."""

from __future__ import annotations

import html
import subprocess
from dataclasses import dataclass, field
from pathlib import Path

import cv2
import numpy as np

from vc_label.contracts import Box
from vc_eval.integrity import FFMPEG

BATCH = 50
GREEN = (0, 200, 0)


def extract_frames(video: Path, frames: list[int], out_dir: Path) -> dict[int, Path]:
    """Decodes 0-based frame numbers of `video` into out_dir/<frame:06d>.jpg."""
    wanted = sorted(set(frames))
    if not wanted:
        return {}
    out_dir.mkdir(parents=True, exist_ok=True)
    got: dict[int, Path] = {}
    for start in range(0, len(wanted), BATCH):
        batch = wanted[start : start + BATCH]
        expr = "+".join(f"eq(n\\,{n})" for n in batch)
        pattern = out_dir / "batch_%05d.jpg"
        subprocess.run(
            [FFMPEG, "-v", "error", "-y", "-i", str(video), "-vf", f"select={expr}", "-fps_mode", "passthrough",
             "-q:v", "3", str(pattern)],
            check=True,
        )
        for i, frame in enumerate(batch, 1):
            produced = out_dir / f"batch_{i:05d}.jpg"
            if not produced.exists():
                raise ValueError(f"{video} has no frame {frame}")
            target = out_dir / f"{frame:06d}.jpg"
            produced.replace(target)
            got[frame] = target
    return got


def _crop(image: np.ndarray, box: Box) -> np.ndarray:
    h, w = image.shape[:2]
    side = max(96.0, 3 * max(box.width, box.height))
    cx, cy = box.center
    x1, y1 = int(max(0, cx - side / 2)), int(max(0, cy - side / 2))
    x2, y2 = int(min(w, cx + side / 2)), int(min(h, cy + side / 2))
    return image[y1:y2, x1:x2].copy(), (x1, y1)


def _to_height(image: np.ndarray, height: int) -> np.ndarray:
    scale = height / image.shape[0]
    return cv2.resize(image, (max(1, round(image.shape[1] * scale)), height), interpolation=cv2.INTER_AREA)


def miss_tile(original: Path, redacted: Path | None, box: Box, out: Path, height: int = 160) -> Path:
    src = cv2.imread(str(original))
    if src is None:
        raise ValueError(f"cannot read {original}")
    crop, (ox, oy) = _crop(src, box)
    cv2.rectangle(crop, (int(box.x1 - ox), int(box.y1 - oy)), (int(box.x2 - ox), int(box.y2 - oy)), GREEN, 1)
    parts = [_to_height(crop, height)]
    if redacted is not None:
        red = cv2.imread(str(redacted))
        if red is None:
            raise ValueError(f"cannot read {redacted}")
        parts.append(_to_height(_crop(red, box)[0], height))
    out.parent.mkdir(parents=True, exist_ok=True)
    cv2.imwrite(str(out), cv2.hconcat(parts), [cv2.IMWRITE_JPEG_QUALITY, 90])
    return out


def thumbnail(image: Path, out: Path, width: int = 640) -> Path:
    src = cv2.imread(str(image))
    if src is None:
        raise ValueError(f"cannot read {image}")
    scale = width / src.shape[1]
    out.parent.mkdir(parents=True, exist_ok=True)
    cv2.imwrite(str(out), cv2.resize(src, (width, round(src.shape[0] * scale)), interpolation=cv2.INTER_AREA))
    return out

class Item:
    image: str  # relative to the page
    caption: str


@dataclass
class Section:
    title: str
    note: str
    items: list[Item]
    links: list[tuple[str, str]] = field(default_factory=list)

h1{font-size:22px;margin:0 0 8px}h2{font-size:17px;margin:28px 0 6px;border-bottom:1px solid #ccc}
.note{color:#555;max-width:72em}.grid{display:flex;flex-wrap:wrap;gap:10px}
figure{margin:0;background:#fff;border:1px solid #ddd;padding:6px;max-width:100%}
figure img{display:block;max-width:100%;height:auto}figcaption{font-size:12px;color:#444;margin-top:4px}
@media (prefers-color-scheme:dark){body{background:#1d1d1f;color:#eee}figure{background:#2a2a2c;border-color:#444}
.note,figcaption{color:#bbb}h2{border-color:#555}a{color:#8ab4f8}}
"""


def write_spotcheck(out_dir: Path, title: str, intro: str, sections: list[Section]) -> Path:
    e = html.escape
    parts = [
        "<!doctype html><html lang=\"zh\"><head><meta charset=\"utf-8\">",
        "<meta name=\"viewport\" content=\"width=device-width,initial-scale=1\">",
        f"<title>{e(title)}</title><style>{_STYLE}</style></head><body>",
        f"<h1>{e(title)}</h1><p class=\"note\">{e(intro)}</p>",
    ]
    for s in sections:
        parts.append(f"<h2>{e(s.title)}</h2><p class=\"note\">{e(s.note)}</p>")
        if s.links:
            parts.append("<p>" + " · ".join(f"<a href=\"{e(href)}\">{e(text)}</a>" for text, href in s.links) + "</p>")
        parts.append("<div class=\"grid\">")
        for item in s.items:
            parts.append(
                f"<figure><img src=\"{e(item.image)}\" loading=\"lazy\" alt=\"{e(item.caption)}\">"
                f"<figcaption>{e(item.caption)}</figcaption></figure>"
            )
        parts.append("</div>")
    parts.append("</body></html>\n")
    out_dir.mkdir(parents=True, exist_ok=True)
    page = out_dir / "index.html"
    page.write_text("".join(parts), encoding="utf-8")
    return page
```

```python
"""Detection-rate sweep over held-out segments, summary tables and the spot-check page (spec §6.3–§6.5)."""

from __future__ import annotations

import json
import random
import subprocess
from dataclasses import dataclass
from pathlib import Path

from vc_label.contracts import LABEL_CLASSES, read_frames_csv
from vc_eval.consistency import compare_detections, python_detections, run_vc_detect
from vc_eval.coverage import CoverageReport, evaluate_coverage
from vc_eval.integrity import IntegrityReport, check_integrity
from vc_eval.metrics import Rate
from vc_eval.spotcheck import Item, Section, extract_frames, miss_tile, thumbnail, write_spotcheck
from vc_eval.temporal import TemporalReport, evaluate_temporal

MAX_MISS_TILES = 60
MAX_TEMPORAL_TILES = 30
STAGES = ("decode", "detect", "track", "mosaic", "encode")


@dataclass
class RunConfig:
    stems: list[str]
    model: Path
    fps_values: list[float]
    out: Path
    vc_redact: Path
    vc_detect: Path
    config: Path
    videos: Path = Path("data/videos")
    frames_root: Path = Path("data/frames")
    labels_root: Path = Path("data/labels")
    redacted_root: Path = Path("data/redacted")
    consistency_count: int = 20
    spot_frames: int = 30
    seed: int = 0


def _fps_dir(cfg: RunConfig, fps: float) -> Path:
    return cfg.redacted_root / f"fps{fps:g}"


def _redact(cfg: RunConfig, stem: str, out: Path, extra: list[str], runner) -> None:
    runner(
        [str(cfg.vc_redact), "--input", str(cfg.videos / f"{stem}.mp4"), "--out", str(out), "--model", str(cfg.model),
         "--config", str(cfg.config), *extra, "--overwrite"],
        check=True, capture_output=True, text=True,
    )


def _rate_json(rate: Rate) -> dict:
    lo, hi = rate.interval
    return {"hits": rate.hits, "total": rate.total, "value": None if rate.total == 0 else rate.value, "ci": [lo, hi]}


def _coverage_json(c: CoverageReport) -> dict:
    out = {cls: _rate_json(c.classes[cls].rate) for cls in LABEL_CLASSES}
    weakest = c.weakest_cell()
    out["weakest_cell"] = None if weakest is None else {"cls": weakest[0], "row": weakest[1], "col": weakest[2], **_rate_json(weakest[3])}
    out["temporal_fill"] = {cls: _rate_json(c.classes[cls].temporal_fill) for cls in LABEL_CLASSES}
    out["best_effort"] = {cls: _rate_json(c.classes[cls].best_effort) for cls in LABEL_CLASSES}
    out["bins"] = {cls: {k: _rate_json(v) for k, v in c.classes[cls].bins.items()} for cls in LABEL_CLASSES}
    out["passed"] = c.passed()
    return out


def _evaluate(cfg: RunConfig, stem: str, fps: float, height: int) -> tuple[dict, CoverageReport, TemporalReport]:
    run_dir = _fps_dir(cfg, fps) / stem
    coverage = evaluate_coverage(cfg.frames_root / stem, cfg.labels_root / stem, run_dir / "masks.jsonl")
    temporal = evaluate_temporal(cfg.redacted_root / "dense" / stem / "detections.jsonl", run_dir / "masks.jsonl", height)
    integrity: IntegrityReport = check_integrity(cfg.videos / f"{stem}.mp4", run_dir / f"{stem}.redacted.mp4")
    stats = json.loads((run_dir / "stats.json").read_text(encoding="utf-8"))
    entry = {
        "stem": stem,
        "fps": fps,
        "coverage": _coverage_json(coverage),
        "temporal": {cls: _rate_json(temporal.classes[cls]) for cls in LABEL_CLASSES},
        "integrity": {"checks": integrity.checks, "details": integrity.details, "passed": integrity.passed},
        "stray": {"max": coverage.stray_max, "mean": coverage.stray_mean, "p95": coverage.stray_p95},
        "stages": stats.get("stages", {}),
        "elapsed_s": stats.get("elapsed_s"),
        "detector": stats.get("detector", ""),
        "encoder": stats.get("encoder", ""),
    }
    return entry, coverage, temporal


def _consistency(cfg: RunConfig, runner) -> dict:
    refs = read_frames_csv(cfg.frames_root / cfg.stems[0])
    step = max(1, len(refs) // max(1, cfg.consistency_count))
    chosen = refs[::step][: cfg.consistency_count]
    cpp = run_vc_detect(cfg.vc_detect, cfg.model, chosen, cfg.out / "consistency", runner=runner)
    report = compare_detections(cpp, python_detections(cfg.model, chosen))
    return {**report.__dict__, "passed": report.passed()}


def _choose(runs: list[dict], fps_values: list[float]) -> float | None:
    for fps in sorted(fps_values):
        mine = [r for r in runs if r["fps"] == fps]
        if mine and all(r["coverage"]["passed"] and r["integrity"]["passed"] for r in mine):
            return fps
    return None


def _spot_sections(cfg: RunConfig, fps: float, reports: dict[str, tuple[CoverageReport, TemporalReport]]) -> list[Section]:
    img = cfg.out / "img"
    rng = random.Random(cfg.seed)
    misses, temporal, random_items, links = [], [], [], []
    for stem, (coverage, tempo) in reports.items():
        video = _fps_dir(cfg, fps) / stem / f"{stem}.redacted.mp4"
        links.append((f"{stem} 打码视频（{fps:g} fps 检测）", Path("..", "..", *video.parts[-4:]).as_posix() if not video.is_absolute() else video.as_uri()))
        chosen_misses = coverage.misses[:MAX_MISS_TILES]
        chosen_temporal = tempo.misses[:MAX_TEMPORAL_TILES]
        pick = sorted(rng.sample(range(coverage.frames * 0 + _frame_count(video)), min(cfg.spot_frames, _frame_count(video))))
        decoded = extract_frames(video, [m.frame for m in chosen_misses] + [m.frame for m in chosen_temporal] + pick, img / stem / "redacted")
        for m in chosen_misses:
            original = cfg.frames_root / stem / f"{m.frame:06d}.jpg"
            tile = miss_tile(original, decoded.get(m.frame), m.box, img / stem / f"miss_{m.frame:06d}_{len(misses)}.jpg")
            misses.append(Item(tile.relative_to(cfg.out).as_posix(), f"{stem} 帧 {m.frame} {m.box.cls} 覆盖 {m.covered:.0%}（左原帧，右打码）"))
        for m in chosen_temporal:
            tile = miss_tile(decoded[m.frame], None, m.box, img / stem / f"temporal_{m.frame:06d}_{len(temporal)}.jpg")
            temporal.append(Item(tile.relative_to(cfg.out).as_posix(), f"{stem} 帧 {m.frame} {m.box.cls} 覆盖 {m.covered:.0%}"))
        for frame in pick:
            thumb = thumbnail(decoded[frame], img / stem / f"random_{frame:06d}.jpg")
            random_items.append(Item(thumb.relative_to(cfg.out).as_posix(), f"{stem} 帧 {frame}"))
    return [
        Section("打码视频", "请完整看一遍，留意是否有露出的人脸或车牌。", [], links),
        Section("未遮住的参照目标", "老师标注里必须遮住、但打码框覆盖不足 80% 的目标。", misses),
        Section("逐帧检测有、稀疏打码漏掉", "同一模型逐帧检测到、但关键帧 + 补框没盖住的框（打码后画面）。", temporal),
        Section("随机打码帧", "随机抽取的打码后画面，检查漏打与误打。", random_items),
    ]


def _frame_count(video: Path) -> int:
    from vc_eval.integrity import probe

    return probe(video).video.frames or 0


def _pct(rate: dict) -> str:
    if not rate["total"]:
        return "—（n=0）"
    lo, hi = rate["ci"]
    return f"{rate['value']:.1%} ({lo:.1%}–{hi:.1%})，n={rate['total']}"


def _summary_md(summary: dict) -> str:
    runs = summary["runs"]
    first = runs[0] if runs else {}
    chosen = summary["chosen_fps"]
    lines = [
        "# 代理验证结果",
        "",
        f"- 模型：{summary['model']}；检测后端：{first.get('detector', '')}；编码器：{first.get('encoder', '')}",
        f"- 选定检测频率：{chosen:g} fps" if chosen is not None else "- 选定检测频率：无满足 P1–P3 的频率",
        "- 本结果以老师融合标注为参照（代理验收），不是人工金标准验收。",
        "",
        "## 最终覆盖率（P1 / P2）",
        "",
        "| 段 | fps | 人脸（95% 区间） | 车牌（95% 区间） | 最弱热力格 | 通过 |",
        "|---|---|---|---|---|---|",
    ]
    for r in runs:
        w = r["coverage"]["weakest_cell"]
        cell = "—" if w is None else f"{w['cls']} r{w['row']}c{w['col']} {w['value']:.1%}（n={w['total']}）"
        lines.append(f"| {r['stem']} | {r['fps']:g} | {_pct(r['coverage']['face'])} | {_pct(r['coverage']['plate'])} | {cell} | {'是' if r['coverage']['passed'] else '否'} |")
    lines += ["", "## 时间覆盖率", "", "| 段 | fps | 人脸 | 车牌 |", "|---|---|---|---|"]
    lines += [f"| {r['stem']} | {r['fps']:g} | {_pct(r['temporal']['face'])} | {_pct(r['temporal']['plate'])} |" for r in runs]
    lines += ["", "## 输出完整性（P3）", "", "| 段 | fps | 帧数 | 时长 | 音频 | 音画同步 |", "|---|---|---|---|---|---|"]
    yes = lambda ok: "✓" if ok else "✗"  # noqa: E731
    for r in runs:
        c = r["integrity"]["checks"]
        lines.append(f"| {r['stem']} | {r['fps']:g} | {yes(c.get('frame_count'))} | {yes(c.get('duration'))} | {yes(c.get('audio_kept'))} | {yes(c.get('av_sync'))} |")
    lines += ["", "## 误打面积（只报告）", "", "| 段 | fps | 最大 | 均值 | 95 分位 |", "|---|---|---|---|---|"]
    lines += [f"| {r['stem']} | {r['fps']:g} | {r['stray']['max']:.2%} | {r['stray']['mean']:.2%} | {r['stray']['p95']:.2%} |" for r in runs]
    lines += ["", "## 速度（本机）", "", "| 段 | fps | 解码 | 检测 | 补框 | 打码 | 编码 | 总耗时 |", "|---|---|---|---|---|---|---|---|"]
    for r in runs:
        st = r["stages"]
        cells = " | ".join(f"{st.get(s, {}).get('fps', 0):.0f}" for s in STAGES)
        lines.append(f"| {r['stem']} | {r['fps']:g} | {cells} | {r['elapsed_s'] or 0:.1f} s |")
    c = summary["consistency"]
    lines += [
        "",
        "## C++ / Python 一致性",
        "",
        f"{c['images']} 张图、{c['boxes']} 个框，配上 {c['matched']} 个，最大坐标差 {c['max_diff_px']:.3f} px，"
        f"未配上（分数 ≥ 0.3）{c['unmatched_confident']} 个、（< 0.3）{c['unmatched_low']} 个：{'通过' if c['passed'] else '未通过'}。",
        "",
    ]
    return "\n".join(lines)


def run_all(cfg: RunConfig, runner=subprocess.run) -> dict:
    cfg.out.mkdir(parents=True, exist_ok=True)
    runs: list[dict] = []
    reports: dict[tuple[str, float], tuple[CoverageReport, TemporalReport]] = {}
    for stem in cfg.stems:
        _redact(cfg, stem, cfg.redacted_root / "dense", ["--detect-every-frame"], runner)
        for fps in cfg.fps_values:
            _redact(cfg, stem, _fps_dir(cfg, fps), ["--detect-fps", f"{fps:g}"], runner)
        height = None
        for fps in cfg.fps_values:
            if height is None:
                from vc_label.review import image_size

                height = image_size(read_frames_csv(cfg.frames_root / stem)[0].image)[1]
            entry, coverage, temporal = _evaluate(cfg, stem, fps, height)
            runs.append(entry)
            reports[(stem, fps)] = (coverage, temporal)
    chosen = _choose(runs, cfg.fps_values)
    summary = {"model": str(cfg.model), "chosen_fps": chosen, "runs": runs, "consistency": _consistency(cfg, runner)}
    spot_fps = chosen if chosen is not None else min(cfg.fps_values)
    sections = _spot_sections(cfg, spot_fps, {stem: reports[(stem, spot_fps)] for stem in cfg.stems})
    write_spotcheck(
        cfg.out,
        "打码抽查",
        f"检测频率 {spot_fps:g} fps。参照为老师融合标注（代理），老师漏标的目标只能靠人眼发现。",
        sections,
    )
    (cfg.out / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    (cfg.out / "summary.md").write_text(_summary_md(summary), encoding="utf-8")
    return summary
```

```python
"""python -m vc_eval coverage | temporal | integrity | run (proxy-validation spec §6)."""

from __future__ import annotations

import argparse
import json
import subprocess
import sys
from pathlib import Path

from vc_label.contracts import LABEL_CLASSES


def _parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="vc_eval")
    sub = p.add_subparsers(dest="command", required=True)
    c = sub.add_parser("coverage", help="final coverage of masks.jsonl against teacher labels")
    c.add_argument("--frames", type=Path, required=True)
    c.add_argument("--labels", type=Path, required=True)
    c.add_argument("--masks", type=Path, required=True)
    c.add_argument("--json", type=Path)
    t = sub.add_parser("temporal", help="sparse masks against a --detect-every-frame run")
    t.add_argument("--dense", type=Path, required=True, help="detections.jsonl of the dense run")
    t.add_argument("--sparse", type=Path, required=True, help="masks.jsonl of the sparse run")
    t.add_argument("--frame-height", type=int, required=True)
    t.add_argument("--json", type=Path)
    i = sub.add_parser("integrity", help="frame count, duration, audio and A/V sync of the output")
    i.add_argument("--source", type=Path, required=True)
    i.add_argument("--redacted", type=Path, required=True)
    i.add_argument("--json", type=Path)
    r = sub.add_parser("run", help="detection-rate sweep, summary and spot-check page")
    r.add_argument("--stems", required=True, help="comma-separated held-out stems")
    r.add_argument("--model", type=Path, required=True)
    r.add_argument("--out", type=Path, required=True)
    r.add_argument("--fps", default="5,10,15,30")
    r.add_argument("--vc-redact", type=Path, default=Path("build/cpp-release/vc_redact"))
    r.add_argument("--vc-detect", type=Path, default=Path("build/cpp-release/vc_detect"))
    r.add_argument("--config", type=Path, default=Path("configs/redact.json"))
    return p


def _rate_text(rate) -> str:
    return f"{rate.hits}/{rate.total}" + (f" ({rate.value:.2%})" if rate.total else "")


def _write_json(path: Path | None, data: dict) -> None:
    if path is not None:
        path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def _coverage(args) -> int:
    from vc_eval.coverage import evaluate_coverage

    r = evaluate_coverage(args.frames, args.labels, args.masks)
    print(f"coverage {r.video}: " + ", ".join(f"{c} {_rate_text(r.classes[c].rate)}" for c in LABEL_CLASSES)
          + f"; {len(r.misses)} misses; {'PASS' if r.passed() else 'FAIL'}")
    _write_json(args.json, {c: {"hits": r.classes[c].rate.hits, "total": r.classes[c].rate.total} for c in LABEL_CLASSES} | {"passed": r.passed()})
    return 0 if r.passed() else 1


def _temporal(args) -> int:
    from vc_eval.temporal import evaluate_temporal

    r = evaluate_temporal(args.dense, args.sparse, args.frame_height)
    print(f"temporal {r.video}: " + ", ".join(f"{c} {_rate_text(r.classes[c])}" for c in LABEL_CLASSES))
    _write_json(args.json, {c: {"hits": r.classes[c].hits, "total": r.classes[c].total} for c in LABEL_CLASSES})
    return 0


def _integrity(args) -> int:
    from vc_eval.integrity import check_integrity

    r = check_integrity(args.source, args.redacted)
    print("integrity: " + ", ".join(f"{k} {'ok' if v else 'FAIL'}" for k, v in r.checks.items()))
    _write_json(args.json, {"checks": r.checks, "details": r.details})
    return 0 if r.passed else 1


def _run(args) -> int:
    from vc_eval.run import RunConfig, run_all

    cfg = RunConfig(
        stems=[s for s in args.stems.split(",") if s], model=args.model, fps_values=[float(v) for v in args.fps.split(",")],
        out=args.out, vc_redact=args.vc_redact, vc_detect=args.vc_detect, config=args.config,
    )
    summary = run_all(cfg)
    chosen = summary["chosen_fps"]
    print(f"run: chosen {chosen:g} fps" if chosen is not None else "run: no detection rate passed P1-P3")
    print(f"run: {args.out / 'index.html'}")
    return 0 if chosen is not None else 1


def main(argv: list[str] | None = None) -> int:
    args = _parser().parse_args(argv)
    try:
        return {"coverage": _coverage, "temporal": _temporal, "integrity": _integrity, "run": _run}[args.command](args)
    except (ValueError, OSError, subprocess.CalledProcessError) as e:
        print(f"vc_eval: {e}", file=sys.stderr)
        return 2
```
